<script>
function calculateVolume() {
    const rawText = document.getElementById('inputText').value.trim();
    const resultsBox = document.getElementById('resultsBox');
    const detailsList = document.getElementById('detailsList');
    
    if (!rawText) {
        alert("Пожалуйста, вставьте текст для расчета.");
        return;
    }
    
    // Меняем запятые между цифрами на точки (30,5 -> 30.5)
    const cleanText = rawText.replace(/(\d+),(\d+)/g, '$1.$2');
    
    // Разбиваем текст на отдельные строки
    const lines = cleanText.split('\n');
    
    let totalVolume = 0.0;
    let totalPieces = 0;
    let detailsHtml = '';
    
    lines.forEach((line, index) => {
        const trimmedLine = line.trim();
        if (!trimmedLine) return; // Пропускаем пустые строки
        
        // Ищем все числа в строке
        const numbers = trimmedLine.match(/\d+(\.\d+)?/g);
        
        if (numbers && numbers.length >= 4) {
            // Берём последние 4 числа
            const exactNumbers = numbers.slice(-4);
            
            const length = parseFloat(exactNumbers[0]);
            const width = parseFloat(exactNumbers[1]);
            const height = parseFloat(exactNumbers[2]);
            const quantity = parseFloat(exactNumbers[3]);
            
            // Расчет объема в м3 (габариты в см)
            const itemVolume = (length * width * height * quantity) / 1000000;
            
            totalVolume += itemVolume;
            totalPieces += quantity;
            
            detailsHtml += `
                <div class="detail-item">
                    <span>Строка ${index + 1}: ${length}x${width}x${height} см × ${quantity} шт.</span>
                    <span style="font-weight:600;">${itemVolume.toFixed(3)} м³</span>
                </div>
            `;
        } else {
            // Если строка не пустая, но чисел мало — выводим предупреждение
            detailsHtml += `
                <div class="detail-item error-item">
                    <span>Строка ${index + 1}: ⚠️ Не нашлось 4 чисел для расчета</span>
                    <span>—</span>
                </div>
            `;
        }
    });
    
    // Выводим результаты на страницу
    document.getElementById('totalVolume').innerHTML = `<strong>${totalVolume.toFixed(3)}</strong> м³`;
    document.getElementById('totalPieces').innerText = `${totalPieces} шт.`;
    detailsList.innerHTML = detailsHtml;
    
    // Показываем блок с результатами
    resultsBox.style.display = 'block';
}
</script>
